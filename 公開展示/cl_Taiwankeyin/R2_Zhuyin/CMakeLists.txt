cmake_minimum_required(VERSION 3.20)
project(cl_Taiwankeyin_R2 VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CELLKEYIN_R1_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../R1_CppCore" CACHE PATH "Verified R1 source root")
if(NOT EXISTS "${CELLKEYIN_R1_ROOT}/src/engine.cpp")
 message(FATAL_ERROR "Verified R1 sibling source not found at ${CELLKEYIN_R1_ROOT}")
endif()
find_package(ICU REQUIRED COMPONENTS uc i18n)
find_package(SQLite3 REQUIRED)
add_library(cellkeyin_core STATIC
 "${CELLKEYIN_R1_ROOT}/src/unicode.cpp" "${CELLKEYIN_R1_ROOT}/src/context.cpp"
 "${CELLKEYIN_R1_ROOT}/src/store.cpp" "${CELLKEYIN_R1_ROOT}/src/engine.cpp")
target_include_directories(cellkeyin_core PUBLIC "${CELLKEYIN_R1_ROOT}/include")
target_link_libraries(cellkeyin_core PUBLIC ICU::uc ICU::i18n SQLite::SQLite3)
add_library(cellkeyin_zhuyin STATIC src/zhuyin.cpp)
target_include_directories(cellkeyin_zhuyin PUBLIC include)
target_link_libraries(cellkeyin_zhuyin PUBLIC cellkeyin_core)
add_executable(cellkeyin_r2_cli src/main.cpp)
target_link_libraries(cellkeyin_r2_cli PRIVATE cellkeyin_zhuyin)
add_executable(cellkeyin_r2_tests tests/test_zhuyin.cpp)
target_link_libraries(cellkeyin_r2_tests PRIVATE cellkeyin_zhuyin)
add_executable(cellkeyin_r1_regression "${CELLKEYIN_R1_ROOT}/tests/test_core.cpp")
target_link_libraries(cellkeyin_r1_regression PRIVATE cellkeyin_core)
if(MSVC)
 foreach(t cellkeyin_core cellkeyin_zhuyin cellkeyin_r2_cli cellkeyin_r2_tests cellkeyin_r1_regression)
  target_compile_options(${t} PRIVATE /W4 /utf-8)
 endforeach()
else()
 foreach(t cellkeyin_core cellkeyin_zhuyin cellkeyin_r2_cli cellkeyin_r2_tests cellkeyin_r1_regression)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endforeach()
endif()
enable_testing()
add_test(NAME R1_Regression COMMAND cellkeyin_r1_regression)
add_test(NAME R2_ZhuyinTests COMMAND cellkeyin_r2_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/seed_zhuyin_tw.tsv")
