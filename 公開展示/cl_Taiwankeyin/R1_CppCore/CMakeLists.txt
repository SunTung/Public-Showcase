cmake_minimum_required(VERSION 3.20)
project(cl_Taiwankeyin VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ICU REQUIRED COMPONENTS uc i18n)
find_package(SQLite3 REQUIRED)
add_library(cellkeyin_core STATIC src/unicode.cpp src/context.cpp src/store.cpp src/engine.cpp)
target_include_directories(cellkeyin_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(cellkeyin_core PUBLIC ICU::uc ICU::i18n SQLite::SQLite3)
if(MSVC)
 target_compile_options(cellkeyin_core PRIVATE /W4 /utf-8)
else()
 target_compile_options(cellkeyin_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(cellkeyin_r1_cli src/main.cpp)
target_link_libraries(cellkeyin_r1_cli PRIVATE cellkeyin_core)
add_executable(cellkeyin_r1_tests tests/test_core.cpp)
target_link_libraries(cellkeyin_r1_tests PRIVATE cellkeyin_core)
enable_testing()
add_test(NAME R1_CoreTests COMMAND cellkeyin_r1_tests)
